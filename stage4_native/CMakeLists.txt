cmake_minimum_required(VERSION 3.20)
project(stage4_peer_identity LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 TEST bridge only")
endif()
set(ELECTRON_HEADERS "" CACHE PATH "Verified matching Electron headers/include/node")
set(ELECTRON_NODE_LIB "" CACHE FILEPATH "Verified matching Electron win-x64/node.lib")
if(NOT EXISTS "${ELECTRON_HEADERS}/node_api.h" OR NOT EXISTS "${ELECTRON_NODE_LIB}")
  message(FATAL_ERROR "Provide local verified headers and import library; no downloads")
endif()
add_library(peer_identity SHARED peer_identity.cc)
target_compile_features(peer_identity PRIVATE cxx_std_17)
target_compile_definitions(peer_identity PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(peer_identity PRIVATE "${ELECTRON_HEADERS}")
target_link_libraries(peer_identity PRIVATE "${ELECTRON_NODE_LIB}" bcrypt delayimp)
target_link_options(peer_identity PRIVATE /DELAYLOAD:node.exe)
set_target_properties(peer_identity PROPERTIES PREFIX "" SUFFIX ".node")
